cmake_minimum_required(VERSION 3.25)
project(CHMOD_Neutral_Examples LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "CHMOD examples require Windows x64")
endif()
foreach(mode observe transform replace)
 add_library(chmod_${mode} SHARED examples/minimal/extension.cpp)
 target_compile_features(chmod_${mode} PRIVATE cxx_std_20)
 target_include_directories(chmod_${mode} PRIVATE include)
 target_compile_definitions(chmod_${mode} PRIVATE CHMOD_EXAMPLE_${mode}=1)
endforeach()
